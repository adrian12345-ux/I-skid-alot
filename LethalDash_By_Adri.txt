local StateTracker = {
	_raw = {
		ScriptEnabled = false,
		SmoothEnabled = false,
		SmoothToggleEnabled = false,
		PanelOpen = false,
		LastCF = Camera.CFrame,
		Busy = false
	}
}

setmetatable(StateTracker, {
	__index = function(t, key)
		return t._raw[key]
	end,
	__newindex = function(t, key, value)
		local old = t._raw[key]
		if old == value then return end
		t._raw[key] = value

		if key == "ScriptEnabled" then
			if value then
				TweenService:Create(toggle, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(55, 5, 55)}):Play()
				TweenService:Create(knob, TweenInfo.new(0.3, Enum.EasingStyle.Quart), {Position = UDim2.fromOffset(40, 2)}):Play()
				label.TextColor3 = Color3.new(1, 1, 1)
			else
				TweenService:Create(toggle, TweenInfo.new(0.3), {BackgroundColor3 = Color3.fromRGB(85, 85, 85)}):Play()
				TweenService:Create(knob, TweenInfo.new(0.3, Enum.EasingStyle.Quart), {Position = UDim2.fromOffset(2, 2)}):Play()
				label.TextColor3 = Color3.fromRGB(85, 85, 85)
				t.SmoothEnabled = false
			end
		elseif key == "SmoothToggleEnabled" then
			if value then
				checkBox.Text = "✓"
				checkBox.BackgroundColor3 = Color3.fromRGB(120, 0, 255)
			else
				checkBox.Text = "X"
				checkBox.BackgroundColor3 = Color3.fromRGB(70, 0, 120)
				t.SmoothEnabled = false
			end
		elseif key == "PanelOpen" then
			if value then
				TweenService:Create(smoothPanel, TweenInfo.new(0.35, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {Size = UDim2.fromOffset(150, 70)}):Play()
				expandBtn.Text = "↑"
			else
				TweenService:Create(smoothPanel, TweenInfo.new(0.25, Enum.EasingStyle.Quart, Enum.EasingDirection.In), {Size = UDim2.fromOffset(150, 0)}):Play()
				expandBtn.Text = "↓"
			end
		end
	end
})

RunService.RenderStepped:Connect(function()
	if not StateTracker.ScriptEnabled or not StateTracker.SmoothEnabled or not StateTracker.SmoothToggleEnabled then return end
	local currentCF = Camera.CFrame  
	StateTracker.LastCF = StateTracker.LastCF:Lerp(currentCF, CONSTANTS.SMOOTH_POWER)  
	Camera.CFrame = StateTracker.LastCF
end)

local Character, Humanoid, HRP
local animConnection

local function getTarget()  
	local liveFolder = workspace:FindFirstChild("Live")  
	if not liveFolder then return nil end  

	local closestHRP = nil  
	local closestDistance = CONSTANTS.LOCK_DISTANCE  

	for _, v in pairs(liveFolder:GetChildren()) do  
		if v:IsA("Model") and v ~= Character then  
			local hrp = v:FindFirstChild("HumanoidRootPart")  
			local humanoid = v:FindFirstChildOfClass("Humanoid")  

			if hrp and humanoid and humanoid.Health > 0 then  
				local distance = (HRP.Position - hrp.Position).Magnitude  
				if distance <= CONSTANTS.LOCK_DISTANCE and distance < closestDistance then  
					closestDistance = distance  
					closestHRP = hrp  
				end  
			end  
		end  
	end  
	return closestHRP  
end  

local function setupCharacter(char)
	Character = char  
	Humanoid = Character:WaitForChild("Humanoid")  
	HRP = Character:WaitForChild("HumanoidRootPart")  
	StateTracker.Busy = false  

	if animConnection then animConnection:Disconnect() end  

	animConnection = Humanoid.AnimationPlayed:Connect(function(track)  
		if not StateTracker.ScriptEnabled then return end  

		if track.Animation and track.Animation.AnimationId == CONSTANTS.DETECT_ANIM then  
			if StateTracker.Busy then return end  
			StateTracker.Busy = true  
			StateTracker.SmoothEnabled = true  
			StateTracker.LastCF = Camera.CFrame  
			
			task.delay(2, function()  
				StateTracker.SmoothEnabled = false  
			end)  

			task.wait(CONSTANTS.WAIT_TIME)  
			if not StateTracker.ScriptEnabled then StateTracker.Busy = false return end  

			Humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
			task.wait()

			local currentVel = HRP.AssemblyLinearVelocity
			HRP.AssemblyLinearVelocity = Vector3.new(currentVel.X, CONSTANTS.VELO_POWER, currentVel.Z)

			Character:WaitForChild("Communicate"):FireServer({  
				Dash = Enum.KeyCode.W,  
				Key = Enum.KeyCode.Q,  
				Goal = "KeyPress"  
			})  

			local target = getTarget()  
			if target then  
				task.delay(0.2, function()  
					local startTime = tick()  
					local connection
					
					connection = RunService.RenderStepped:Connect(function()  
						if not StateTracker.ScriptEnabled or (tick() - startTime) >= CONSTANTS.SNAP_DURATION or not target or not target.Parent then  
							Humanoid.AutoRotate = true  
							connection:Disconnect()  
							return  
						end  

						Humanoid.AutoRotate = false  
						local myPos = HRP.Position  
						local targetPos = target.Position  

						local flatTarget = Vector3.new(targetPos.X, myPos.Y, targetPos.Z)  
						local targetCF = CFrame.new(myPos, flatTarget)  

						local progress = math.clamp((tick() - startTime) / CONSTANTS.SNAP_DURATION, CONSTANTS.LERP_START_ALPHA, CONSTANTS.LERP_END_ALPHA)
						local smooth = progress * progress * (3 - 2 * progress)

						HRP.CFrame = HRP.CFrame:Lerp(targetCF, smooth)
					end)
				end)
			end
			StateTracker.Busy = false  
		end  
	end)
end

if LocalPlayer.Character then setupCharacter(LocalPlayer.Character) end

toggle.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
		StateTracker.ScriptEnabled = not StateTracker.ScriptEnabled
	end
end)

local dragging = false
local dragInput, dragStart, startPos

main.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
		if dragging then return end  
		dragging = true  
		dragInput = input  
		dragStart = input.Position  
		startPos = main.Position  
	end
end)

UIS.InputChanged:Connect(function(input)
	if dragging and input == dragInput then
		local delta = input.Position - dragStart
		main.Position = UDim2.new(
			startPos.X.Scale,
			startPos.X.Offset + delta.X,
			startPos.Y.Scale,
			startPos.Y.Offset + delta.Y
		)
	end
end)

UIS.InputEnded:Connect(function(input)
	if input == dragInput then
		dragging = false
		dragInput = nil
	end
end)
